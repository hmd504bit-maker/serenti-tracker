function doPost(e) { var raw = e.postData.contents; var decoded = decodeURIComponent(raw.replace(/^data=/, '')); decoded = decoded.replace(/\u201c|\u201d|\u300c|\u300d|"/g, '"'); decoded = decoded.replace(/'/g, "'"); var jsonMatch = decoded.match(/\[\s*\{[\s\S]*?\}\s*\]/); var postMatch = decoded.match(/POST_START\s*([\s\S]*?)\s*POST_END/); if (!jsonMatch) { return ContentService.createTextOutput('投稿文なし'); } var jsonStr = jsonMatch[0].replace(/\+\+/g, '').trim(); try { var stocks = JSON.parse(jsonStr); } catch(err) { return ContentService.createTextOutput('投稿文なし'); } var postText = postMatch ? postMatch[1].trim() : ''; if (postText.length > 140) { postText = postText.substring(0, 137) + '...'; } var ss = SpreadsheetApp.openById('18D1RgxG9blhUytKSzUAqBdA-_wftW8lFX5AAET9P3YI'); var sheet = ss.getActiveSheet(); var data = sheet.getDataRange().getValues(); var finnhubKey = 'd7s4rfhr01qm28g8b1r0d7s4rfhr01qm28g8b1rg'; for (var j = 0; j < stocks.length; j++) { var stock = stocks[j]; // ティッカーを必ず文字列に変換 var ticker = String(stock.ticker).trim(); if (!ticker || ticker === 'undefined') continue; var found = false; for (var i = 1; i < data.length; i++) { // スプレッドシート側も文字列に変換して比較 if (String(data[i][1]).trim() === ticker) { found = true; var prevSentiment = data[i][2]; var newSentiment = stock.sentiment; var currentPrice = data[i][4]; sheet.getRange(i+1, 6).setValue(prevSentiment); sheet.getRange(i+1, 3).setValue(newSentiment); sheet.getRange(i+1, 4).setValue(Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd')); sheet.getRange(i+1, 7).setValue(Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd')); if (!currentPrice || currentPrice === 0 || prevSentiment !== newSentiment) { try { var url = 'https://finnhub.io/api/v1/quote?symbol=' + ticker + '&token=' + finnhubKey; var response = UrlFetchApp.fetch(url); var quote = JSON.parse(response.getContentText()); if (quote.c && quote.c > 0) { sheet.getRange(i+1, 5).setValue(quote.c); sheet.getRange(i+1, 8).setValue(Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd')); } } catch(err) {} } break; } } if (!found) { var newRow = [ stock.name || '', ticker, stock.sentiment || '', Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd'), 0, '', Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd'), '' ]; sheet.appendRow(newRow); try { var url = 'https://finnhub.io/api/v1/quote?symbol=' + ticker + '&token=' + finnhubKey; var response = UrlFetchApp.fetch(url); var quote = JSON.parse(response.getContentText()); if (quote.c && quote.c > 0) { var lastRow = sheet.getLastRow(); sheet.getRange(lastRow, 5).setValue(quote.c); sheet.getRange(lastRow, 8).setValue(Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd')); } } catch(err) {} } } return ContentService.createTextOutput(postText); } function doGet(e) { var ss = SpreadsheetApp.openById('18D1RgxG9blhUytKSzUAqBdA-_wftW8lFX5AAET9P3YI'); var sheet = ss.getActiveSheet(); var data = sheet.getDataRange().getValues(); var stocks = []; for (var i = 1; i < data.length; i++) { var ticker = String(data[i][1]).trim(); if (ticker && ticker !== 'undefined') { stocks.push({ name: data[i][0], ticker: ticker, sentiment: data[i][2] || 'neu', date: data[i][3] || '', basePrice: data[i][4] || 0, prevSentiment: data[i][5] || '', updatedAt: data[i][6] || '', basePriceDate: data[i][7] || '' }); } } var output = ContentService.createTextOutput(JSON.stringify(stocks)); output.setMimeType(ContentService.MimeType.JSON); return output; }
